rules_version = '2';

// ============================================================
//  REGLAS DE SEGURIDAD - OFICINA DIGITAL
// ------------------------------------------------------------
//  AL INSTALAR UNA OFICINA NUEVA:
//  cambia el correo de la linea marcada como CORREO DEL PANEL
//  por el del distribuidor. Es lo UNICO que se cambia aqui.
//
//  Si te equivocas en ese correo, el distribuidor no podra
//  entrar a su panel y tu si podrias ver sus datos. Revisalo.
// ============================================================

service cloud.firestore {
  match /databases/{database}/documents {
    // Solo el distribuidor puede editar y borrar.
    function esAdmin() {
      return request.auth != null
        // >>> CAMBIAR POR EL CORREO DEL DISTRIBUIDOR <<<
        && request.auth.token.email == 'CORREO-DEL-DISTRIBUIDOR@gmail.com';
    }
    // Recetas, galeria y proximamente: lectura publica, escritura solo el distribuidor.
    match /recipes/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    match /gallery/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    match /upcoming/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    // Opiniones: lectura publica. el distribuidor puede todo.
    // Los visitantes pueden ENVIAR una resena nueva (que llega oculta y pendiente),
    // pero no pueden editar ni borrar las existentes.
    match /testimonials/{id} {
      allow read: if true;
      allow update, delete: if esAdmin();
      allow create: if esAdmin() || (
        request.resource.data.status == 'hidden'
        && request.resource.data.visible == false
        && request.resource.data.pendiente == true
        && request.resource.data.stars is int
        && request.resource.data.stars >= 1
        && request.resource.data.stars <= 5
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && request.resource.data.name.size() < 60
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() < 800
        && (!('photo' in request.resource.data)
            || (request.resource.data.photo is string
                && request.resource.data.photo.size() < 700000
                && request.resource.data.photo.matches('data:image/.*')))
      );
    }
    // Analitica: cualquiera registra visitas y clics, solo el distribuidor lee.
    match /analytics_clicks/{docId} {
      allow read: if esAdmin();
      allow create: if request.resource.data.label is string
        && request.resource.data.label.size() < 100;
      allow update, delete: if false;
    }
    // Contadores publicos de recetas: cualquiera lee, cualquiera suma o resta
    // de uno en uno. No guarda ningun dato personal, solo numeros.
    match /recipeStats/{recipeId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['likes','saves'])
        && request.resource.data.likes is int && request.resource.data.likes >= 0 && request.resource.data.likes <= 1
        && request.resource.data.saves is int && request.resource.data.saves >= 0 && request.resource.data.saves <= 1;
      allow update: if request.resource.data.keys().hasOnly(['likes','saves'])
        && request.resource.data.likes is int && request.resource.data.saves is int
        && request.resource.data.likes >= 0 && request.resource.data.saves >= 0
        && (request.resource.data.likes - resource.data.likes) >= -1
        && (request.resource.data.likes - resource.data.likes) <= 1
        && (request.resource.data.saves - resource.data.saves) >= -1
        && (request.resource.data.saves - resource.data.saves) <= 1;
      allow delete: if esAdmin();
    }
    // Marca anonima de "me gusta" para que un mismo navegador no cuente dos veces.
    // El identificador va en el nombre del documento y no identifica a nadie.
    match /recipeLikes/{docId} {
      allow get: if true;
      allow list: if esAdmin();
      allow create: if !exists(/databases/$(database)/documents/recipeLikes/$(docId))
        && request.resource.data.keys().hasOnly(['recipeId'])
        && request.resource.data.recipeId is string
        && request.resource.data.recipeId.size() < 80;
      allow delete: if true;
      allow update: if false;
    }
    // Centro de ayuda: cualquiera puede ENVIAR una solicitud, solo el distribuidor la lee.
    // Nadie puede leer, listar ni modificar solicitudes ajenas.
    match /helpRequests/{docId} {
      allow read, list: if esAdmin();
      allow update, delete: if esAdmin();
      allow create: if request.resource.data.tipo is string
        && request.resource.data.tipo in ['producto','servicio','garantia','llamada','visita']
        && request.resource.data.nombre is string
        && request.resource.data.nombre.size() > 0
        && request.resource.data.nombre.size() < 80
        && request.resource.data.apellido is string
        && request.resource.data.apellido.size() < 80
        && request.resource.data.phone is string
        && request.resource.data.phone.size() >= 7
        && request.resource.data.phone.size() < 30
        && request.resource.data.consent == true
        && request.resource.data.status == 'nueva'
        && (!('detalle' in request.resource.data) || request.resource.data.detalle.size() < 2000)
        && (!('photo' in request.resource.data)
            || (request.resource.data.photo is string
                && request.resource.data.photo.size() < 700000
                && request.resource.data.photo.matches('data:image/.*')));
    }
    // Fichas de cliente: se construyen agrupando las solicitudes recibidas.
    // Nadie del publico escribe aqui; solo el distribuidor guarda notas y datos internos.
    match /clients/{docId} {
      allow read, write: if esAdmin();
    }
    // Papelera (30 dias): solo el distribuidor.
    match /trash/{docId} {
      allow read, write: if esAdmin();
    }
    // Configuracion publica: lectura publica, escritura solo el distribuidor.
    match /settings/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    // Pasaporte de Sabores: cada viajero (anonimo) solo toca SU documento.
    match /passportUsers/{uid} {
      allow read: if esAdmin() || (request.auth != null && request.auth.uid == uid);
      allow create: if request.auth != null && request.auth.uid == uid
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() < 50
        && request.resource.data.phone is string
        && request.resource.data.phone.size() < 25
        && request.resource.data.city is string
        && request.resource.data.city.size() < 50
        && request.resource.data.state is string
        && request.resource.data.state.size() < 40
        && request.resource.data.consent == true
        && request.resource.data.stamps is map
        && request.resource.data.stamps.size() <= 60;
      allow update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.name is string
        && request.resource.data.name.size() < 50
        && request.resource.data.phone is string
        && request.resource.data.phone.size() < 25
        && request.resource.data.stamps is map
        && request.resource.data.stamps.size() <= 60
        && request.resource.data.stamps.keys().hasAll(resource.data.stamps.keys());
      allow delete: if esAdmin();
    }
    // Fotos del pasaporte: cada viajero sube las suyas; solo el distribuidor las ve todas.
    match /passportPhotos/{docId} {
      allow read: if esAdmin() || (request.auth != null && resource.data.uid == request.auth.uid);
      allow create, update: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && docId == request.auth.uid + '__' + request.resource.data.country
        && request.resource.data.country is string
        && request.resource.data.country.size() < 40
        && request.resource.data.photo is string
        && request.resource.data.photo.size() < 700000
        && request.resource.data.photo.matches('data:image/.*')
        && request.resource.data.name is string
        && request.resource.data.name.size() < 50;
      allow delete: if esAdmin();
    }

    // ============================================================
    // REGALA Y GANA (nuevo - no afecta nada de lo anterior)
    // ============================================================

    function telNormalValido(t) {
      return t is string && t.size() >= 10 && t.size() <= 15
        && t.matches('[0-9]+');
    }
    function textoValido(t, min, max) {
      return t is string && t.size() >= min && t.size() < max;
    }

    // Enlace publico del recomendador: solo nombre, ciudad y estado.
    match /giftLinks/{code} {
      allow read: if true;
      allow create: if request.auth != null
        && code.size() >= 10 && code.size() <= 14
        && code.matches('[A-Z0-9]+')
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.status == 'active'
        && textoValido(request.resource.data.participantName, 2, 60)
        && textoValido(request.resource.data.city, 2, 60)
        && textoValido(request.resource.data.distributorId, 2, 40)
        && request.resource.data.startsAt is timestamp
        && request.resource.data.expiresAt is timestamp;
      allow update, delete: if esAdmin();
    }

    // Datos privados del participante: solo su dueño (anonimo) y el distribuidor.
    // El participante NO puede modificarse despues de crearse:
    // sus contadores los escribe unicamente el administrador.
    match /giftParticipants/{code} {
      allow read: if esAdmin()
        || (request.auth != null && resource.data.ownerUid == request.auth.uid);
      allow create: if request.auth != null
        && code.size() >= 10 && code.size() <= 14
        && code.matches('[A-Z0-9]+')
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.consentTerms == true
        && textoValido(request.resource.data.participantName, 2, 60)
        && textoValido(request.resource.data.city, 2, 60)
        && textoValido(request.resource.data.distributorId, 2, 40)
        && request.resource.data.status == 'active'
        && request.resource.data.giftsDelivered == 0
        && request.resource.data.salesConfirmed == 0
        && request.resource.data.claimsTotal == 0;
      allow update, delete: if esAdmin();
    }

    // Reclamos: cualquiera con sesion anonima puede CREAR uno,
    // con ID = CODIGO__TELEFONO (evita duplicados de forma atomica)
    // y solo si el enlace existe y sigue activo. Nadie mas los lee.
    match /giftClaims/{claimId} {
      allow read: if esAdmin();
      allow create: if request.auth != null
        && claimId == request.resource.data.referralCode + '__' + request.resource.data.normalizedPhone
        && exists(/databases/$(database)/documents/giftLinks/$(request.resource.data.referralCode))
        && get(/databases/$(database)/documents/giftLinks/$(request.resource.data.referralCode)).data.status == 'active'
        && request.resource.data.distributorId == get(/databases/$(database)/documents/giftLinks/$(request.resource.data.referralCode)).data.distributorId
        && request.resource.data.participantId == request.resource.data.referralCode
        && request.resource.data.status == 'claimed'
        && request.resource.data.consent == true
        && telNormalValido(request.resource.data.normalizedPhone)
        && textoValido(request.resource.data.recipientName, 2, 80)
        && textoValido(request.resource.data.recipientCity, 2, 60)
        && (!('zipCode' in request.resource.data)
            || (request.resource.data.zipCode is string
                && request.resource.data.zipCode.size() < 12));
      allow update, delete: if esAdmin();
    }

    // Escala de premios editable: todos la ven, solo el distribuidor la cambia.
    match /rewardLevels/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }

    // Recompensas otorgadas: solo el distribuidor.
    match /rewardRedemptions/{id} {
      allow read, write: if esAdmin();
    }

    // Auditoria: solo el distribuidor crea y lee. NADIE edita ni borra jamas.
    match /auditLogs/{id} {
      allow read: if esAdmin();
      allow create: if esAdmin();
      allow update, delete: if false;
    }

    // Todo lo demas: privado.
    match /{document=**} {
      allow read, write: if esAdmin();
    }
  }
}
